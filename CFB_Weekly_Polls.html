let activeSeason = "2026-27";


function esc(s) {

  return String(s).replace(/[&<>"']/g, m => ({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[m]));

}


function movementHtml(move) {

  if (!move) return "";

  if (move === "new") return '<span class="move new">NEW</span>';

  if (move === "x") return '<span class="move unchanged">×</span>';

  const n = parseInt(move,10);

  if (Number.isNaN(n)) return '<span class="move unchanged">' + esc(move) + '</span>';

  return '<span class="move ' + (n > 0 ? 'up' : 'down') + '">(' + (n > 0 ? '+' : '') + n + ')</span>';

}


function renderTabs() {

  const tabs = document.getElementById("tabs");

  tabs.innerHTML = Object.keys(DATA.seasons).map(year =>

    '<button class="tab ' + (year === activeSeason ? 'active' : '') + '" data-year="' + year + '">' + year + '</button>'

  ).join("");

  tabs.querySelectorAll(".tab").forEach(btn => btn.addEventListener("click", () => {

    activeSeason = btn.dataset.year;

    render();

  }));

}


function renderPoll(season) {

  const weeks = season.weeks;

  const latestIdx = weeks.length - 1;

  let html = '<div class="poll-grid">';

  html += '<div class="rank-col header-row rank-head">Rank</div>';

  for (let r = 0; r < 25; r++) {

    html += '<div class="rank-col rank-row"><span class="rank-num">' + (r+1) + '</span></div>';

  }

  weeks.forEach((w, i) => {

    html += '<div class="grid-cell header-row week-head ' + (i === latestIdx ? 'latest' : '') + '"><div class="date">' + esc(w.date) + '</div><div class="label">' + esc(w.label) + '</div></div>';

    for (let r = 0; r < 25; r++) {

      const item = w.entries[r];

      html += '<div class="grid-cell team-cell ' + (i === latestIdx ? 'latest' : '') + '"><span class="team-name">' + esc(item.team) + '</span>' + movementHtml(item.move) + '</div>';

    }

  });

  html += '</div>';

  document.getElementById("pollScroll").innerHTML = html;

  const scroller = document.getElementById("pollScroll");

  requestAnimationFrame(() => {

    scroller.scrollLeft = scroller.scrollWidth;

  });

}


function numbered(items) {

  if (!items || !items.length) return "";

  return '<ol>' + items.map(x => '<li>' + esc(x) + '</li>').join("") + '</ol>';

}


function renderSupplemental(season) {

  const mount = document.getElementById("supplemental");

  const cards = [];

  if (season.sec_qb) cards.push('<article class="card"><h3>26-27 SEC QB Rankings</h3>' + numbered(season.sec_qb) + '</article>');

  if (season.predictions) cards.push('<article class="card"><h3>Pre-Season Predictions</h3>' + numbered(season.predictions) + '</article>');

  if (season.cfp) cards.push('<article class="card"><h3>CFP 27 Prediction</h3>' + numbered(season.cfp) + '</article>');

  mount.innerHTML = cards.length ? '<div class="cards">' + cards.join("") + '</div>' : "";

}


function render() {

  const season = DATA.seasons[activeSeason];

  renderTabs();

  document.getElementById("currentSeason").textContent = activeSeason;

  document.getElementById("pollTitle").textContent = activeSeason + " — " + season.subtitle;

  document.getElementById("pollHint").textContent = season.weeks.length + " weekly columns • fixed 1–25 ranking rows";

  renderPoll(season);

  renderSupplemental(season);

  document.title = "Mehtaverse CFB Poll — " + activeSeason;

}
